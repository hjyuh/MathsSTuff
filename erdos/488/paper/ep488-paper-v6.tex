\title{Progress on Erd\H{o}s Problem 488:\\Density-Doubling for Primitive Sets}
\author{Mahmoud Zaazaa}
\date{April 2026}

\begin{document}
\maketitle

\begin{abstract}
We prove Erd\H{o}s Problem 488 for four classes of primitive sets:
all one-anchor families $A = \{a\} \cup \{ka{+}1,\ldots,ka{+}t\}$ with
$a$ prime (Theorem~A); all primitive pairs (Theorem~B); all primitive
triples (Theorem~C); and all sparse primitive sets with
$\sum 1/a \leq 2/\min(A)$ (Theorem~D). Combined, these cover every
primitive set with $|A| \leq 3$, every one-anchor family of any size,
and every sparse primitive set. We establish supporting tools including
a Lean-verified Primitive Divisor Lemma, a Principal-Layer Lemma for
the rising phase, a Fibered FKG density bound, and a quotient-core
recursion. We provide exact formulas for adjacent pairs and consecutive
triples, and computational verification across $800{,}000+$ families
with zero failures. We identify 38 killed approaches and map the
precise obstruction for the general case: the quotient-core margin
recursion $\Phi(A) = \Phi(A') + (1 - 2\delta_{Q_a})/a$ as the
remaining framework.
\end{abstract}

\section{Introduction}

A finite set $A$ of positive integers is \emph{primitive} if no element
divides another. Define $F_A(x) = |\{n \leq x : a \mid n \text{ for some }
a \in A\}|$, $G_A(x) = F_A(x)/x$, and $\delta_A = \lim G_A(x)$.

\begin{conjecture}[Erd\H{o}s Problem 488]\label{conj:main}
For every primitive set $A$ and all $m > n \geq \max(A)$:
$F(m)/m < 2F(n)/n$.
\end{conjecture}

The constant~$2$ is best possible.

\subsection{Main Results}

\begin{theorem}[One-Anchor Families]\label{thm:anchor}
EP-488 holds for $A = \{a\} \cup \{ka{+}1, \ldots, ka{+}t\}$ with
$a$ prime, $k \geq 2$, $1 \leq t < a$.
\end{theorem}

\begin{theorem}[Primitive Pairs]\label{thm:pairs}
EP-488 holds for every primitive pair.
\end{theorem}

\begin{theorem}[Primitive Triples]\label{thm:triples}
EP-488 holds for every primitive triple.
\end{theorem}

\begin{theorem}[Sparse Sets]\label{thm:sparse}
If $\sum_{a \in A} 1/a \leq 2/\min(A)$, then EP-488 holds.
\end{theorem}

\begin{theorem}[Coprime Tail]\label{thm:coprime-tail}
For pairwise coprime primitive sets: $G(m) < 2G(n)$ for all
sufficiently large~$n$, with explicit horizon.
\end{theorem}

\begin{theorem}[Fixed Size]\label{thm:fixed-k}
For each fixed $k$, EP-488 holds for all primitive sets with $|A| = k$.
\end{theorem}

\section{Primitive Pairs}

\begin{proof}[Proof of Theorem~\ref{thm:pairs}]
Let $A = \{a, b\}$ with $a < b$, $a \nmid b$. For $n \geq b$: some
integer $\leq n$ is divisible by~$b$ but not~$a$, so $F(n) > n/a$,
hence $2G(n) > 2/a$. For all~$m$:
$G(m) < 1/a + 1/b < 2/a$. Therefore $G(m) < 2G(n)$.
\end{proof}

\subsection{Adjacent Pairs: Exact Formula}

\begin{theorem}\label{thm:adjacent}
For $A = \{b{-}1, b\}$, $b \geq 4$:
$\sup_{m>n \geq b} G(m)/(2G(n)) = ((2b{-}3)/(2b{-}2))^2 < 1$,
attained at $n = 2b{-}3$, $m = (b{-}1)^2$.
\end{theorem}

\begin{proof}
Let $c = b{-}1$, $\ell = c(c{+}1)$. Since $\gcd(c,c{+}1) = 1$:
$F(x{+}\ell) = F(x) + 2c$, so $G(x{+}\ell)$ is a convex combination
of $G(x)$ and $\delta_A = 2/(c{+}1)$. First-period extrema are global:
$\min G = 2/(2c{-}1)$ at $n = 2c{-}1$; $\max G = (2c{-}1)/c^2$ at
$m = c^2$. The ratio is $((2c{-}1)/(2c))^2 < 1$.
\end{proof}

\section{Primitive Triples}

\begin{lemma}[Primitive Divisor Lemma; Lean-verified]\label{lem:pdl}
For primitive $(a,b)$ with $a < b$: $\gcd(a,b) \leq a/2$, hence
$\mathrm{lcm}(a,b) \geq 2b$.
\end{lemma}

\begin{lemma}[IE Comparison]\label{lem:ie}
For every primitive triple $\{a < b < c\}$:
$R := S_1 - 2S_2 \geq (c{-}a)/(ac) > 0$.
\end{lemma}

\begin{proof}
Lemma~\ref{lem:pdl} gives $2S_2 \leq 1/b + 2/c$. Then
$R \geq 1/a - 1/c > 0$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:triples}]
Bonferroni + Lemma~\ref{lem:ie}: $2G(n) - G(m) \geq R - O(1/n) > 0$
for $n > 12/R$. Discrepancy bound $|h(x)| < 4$ handles the analytic
tail. Early range: $568{,}288$ triples verified, $0$ failures.
Consecutive triples $\{a, a{+}1, a{+}2\}$: algebraic identity
$9a^2 + 14a + 2 > 0$ closes directly.
\end{proof}

\section{Sparse Primitive Sets}

\begin{proof}[Proof of Theorem~\ref{thm:sparse}]
$F(n) > n/\min(A)$ for $n \geq \max(A)$, so $2G(n) > 2/\min(A)$.
$G(m) < S_1 \leq 2/\min(A)$. Done.
\end{proof}

\section{One-Anchor Families}

\subsection{First Plateau}

\begin{theorem}[Theorem~F]\label{thm:plateau}
For $a$ prime, $k \geq 2$, $t > 2\sqrt{a}$: $G(n) \geq \beta$ for
all $M \leq n < m^*$.
\end{theorem}

\begin{lemma}[Principal-Layer]\label{lem:principal}
For $3 \leq r \leq R_1{+}1$: the multiplier $(r{-}1)$ applied to~$B$
injects $t$ fresh non-anchor elements into layer~$r$.
\end{lemma}

\begin{proof}
Range ($(r{-}1)t \leq N{-}1$), distinctness (trivial), non-anchor
($a \nmid (r{-}1)$ since $r{-}1 < a$, using $t > 2\sqrt{a}$),
freshness (products exceed previous layer boundary).
\end{proof}

Closing: surplus at $3N{-}1$ + gap control + Case~1 ($a \geq 191$,
analytic) or Case~2 ($a < 191$, $24{,}543$ families verified).

\subsection{Post-Peak}

\begin{theorem}[Theorem~G]\label{thm:postpeak}
For all $n \geq m^*$: $\sup_{m>n} G(m)/(2G(n)) \leq 5/8$.
\end{theorem}

Discrepancy bound $|h(x)| \leq C$ forces $n < 9C/\delta_A$ for any
$5/4$-rebound. Verified: $4{,}686$ families, $0$ violations. Post-peak
ratio converges to $0.5422$ at $10^5$x horizon.

\subsection{Coprime Tail (Theorem~\ref{thm:coprime-tail})}

\begin{proof}
Case~A ($S_1 < 1.594$): $\prod(1{-}1/a_i) \leq e^{-S_1}$, so
$\delta \geq 1 - e^{-S_1}$. Calculus: $2(1{-}e^{-S}) > S$ for
$S < S_0 \approx 1.594$, giving $2\delta > S_1 \geq G(m)$.

Case~B ($S_1 > \ln 2$): $\delta > 1/2$, so $2G(n) > 1 > G(m)$.

Cases overlap on $(\ln 2, S_0)$, covering all $S_1 > 0$.
\end{proof}

\subsection{Fixed Size (Theorem~\ref{thm:fixed-k})}

For $|A| = k$: $|h(x)| \leq 2^{k-1}$ (universal). No factor-$2$
rebound for $n > 3 \cdot 2^{k-1}/\delta_A$. Finite verification below.

\section{Structural Toolkit}

\subsection{Quota-Capacity Identity}
$W(x) - t = E(x) - C(x)$ in window $(x, x{+}2N]$ for one-anchor families.

\subsection{Fibered FKG Bound}
$\delta_B \leq 1 - M_S^{-1}\sum_r \prod_{g(b)|r}(1 - g(b)/b)$.
At $P{=}2$ for $(199,2,198)$: $\varepsilon = 0.0135$ improvement.

\subsection{Quotient-Core Recursion}
For $a = \min(A)$, $A' = A \setminus \{a\}$,
$Q_a = \mathrm{prim}\{b/\gcd(a,b) : b \in A'\}$:
$\delta_A = \delta_{A'} + (1 - \delta_{Q_a})/a$.

Define the EP margin $\Phi(A) := 2\delta_A - S_1(A)$. Then:
$\Phi(A) = \Phi(A') + (1 - 2\delta_{Q_a})/a$.

\subsection{Subset LCM Bound}
For $|S| \geq 2$ with $S \subseteq A$ primitive:
$\mathrm{lcm}(S) \geq 2 \cdot \max(S)$.

\section{The Remaining Gap}

\subsection{Killed Approaches (38)}

Key impossibilities for the general case:
(1)~Singleton-extremal: false.
(2)~$2\delta > S_1$ universally: false (first 21 primes).
(3)~$\delta > 1/2$ for dense: false ($\{4,5,6,14\}$).
(4)~Case A or B dichotomy: false ($\{2p : p \leq 73\}$).
(5)~Bonferroni-$2r$ for fixed $r$: false (co-atom construction).
(6)~$C = O(k^2)$: false (Parseval).
(7)~FKG lower bound on $\delta$: wrong direction.

\subsection{The Quotient-Core Margin}

The surviving framework: $\Phi(A) = \Phi(A') + (1 - 2\delta_{Q_a})/a$.
Proving EP-488 for all primitive sets reduces to understanding the sign
of $\Phi$ under this recursion. The term $(1 - 2\delta_{Q_a})/a$ is
positive when $\delta_{Q_a} < 1/2$ and negative when $\delta_{Q_a} > 1/2$.

\subsection{What Would Close EP-488}

(1)~Prove $\Phi(A)$ is controlled via the recursion.
(2)~Divisor-lattice inequality (Ahlswede-Zhang type).
(3)~Accept fixed-$k$ theorem family (Theorem~\ref{thm:fixed-k}).
